def add_v381_features(self, df: pd.DataFrame, track_name: str) -> pd.DataFrame:
    """添加 v3.8.1 特徵（18個 - v3.8 + driver_track_affinity）"""
    df = df.copy()
    
    # ========== v3.3 交互特徵 (3) ==========
    df['s1_s2_ratio'] = df['ideal_s1'] / (df['ideal_s2'] + 1e-6)
    sector_mean = (df['ideal_s1'] + df['ideal_s2'] + df['ideal_s3']) / 3
    sector_std = df[['ideal_s1', 'ideal_s2', 'ideal_s3']].std(axis=1)
    df['sector_cv'] = sector_std / (sector_mean + 1e-6)
    df['s2_lap_ratio'] = df['ideal_s2'] / (df['ideal_lap'] + 1e-6)
    
    # ========== v3.4 速度特徵 (3) ==========
    df['max_speed_lap_ratio'] = df['max_speed'] / (df['ideal_lap'] + 1e-6)
    df['max_speed_s2_ratio'] = df['max_speed'] / (df['ideal_s2'] + 1e-6)
    apex_speeds = df[['low_speed_apex', 'mid_speed_apex', 'high_speed_apex']]
    df['speed_consistency'] = apex_speeds.std(axis=1) / (df['max_speed'] + 1e-6)
    
    # ========== v3.5 有效特徵 (3) ==========
    df['fp3_relative_position'] = df['ideal_lap'].rank(method='min')
    df['fp3_gap_to_fastest'] = df['ideal_lap'] - df['ideal_lap'].min()
    df['is_top_driver'] = df['driver'].isin(self.top_drivers).astype(int)
    
    # ========== v3.8.1 論文特徵 (1) ✨ 新增 ==========
    # 特徵 18: driver_track_affinity (車手×賽道交互)
    # 論文: driver_name:circuit_name 交互項 (p < 0.05 統計顯著)
    
    # 車手編碼
    if not self.driver_encoder_fitted:
        # 首次 fit：使用所有車手
        all_drivers = df['driver'].unique()
        self.driver_encoder.fit(all_drivers)
        self.driver_encoder_fitted = True
        print(f"  [Encoder] 車手編碼器已 fit，共 {len(all_drivers)} 位車手")
    
    # 處理未知車手 (新車手)
    known_drivers = set(self.driver_encoder.classes_)
    df['driver_encoded'] = df['driver'].apply(
        lambda x: self.driver_encoder.transform([x])[0] if x in known_drivers else -1
    )
    
    # 賽道編碼（使用 hash，確保不同賽道有不同編碼）
    track_hash = hash(track_name) % 1000  # 映射到 0-999
    
    # 交互項 = 車手編碼 × 賽道編碼
    # 未知車手的交互項設為 0
    df['driver_track_affinity'] = df['driver_encoded'].apply(
        lambda x: x * track_hash if x != -1 else 0
    )
    
    # 移除臨時列
    df = df.drop(columns=['driver_encoded'])
    
    print(f"  [Feature] driver_track_affinity 範圍: [{df['driver_track_affinity'].min():.0f}, {df['driver_track_affinity'].max():.0f}]")
    
    return df
```

def test_v381_features():

        'fp3_relative_position', 'fp3_gap_to_fastest', 'is_top_driver',
        # v3.8.1 論文特徵 (1) ✨
        'driver_track_affinity'

# compare_v38_v381_performance.py

#!/usr/bin/env python3
"""
v3.8 vs v3.8.1 性能比較

比較項目：
1. CV MAE
2. Spearman Correlation
3. Top5 Accuracy
4. 模型大小
5. 訓練時間
"""

def compare_versions():
    """比較 v3.8 和 v3.8.1"""
    
    # 載入 v3.8 結果
    with open('models/track_specific_v3.8_summary.json', 'r') as f:
        v38_results = json.load(f)
    
    # 載入 v3.8.1 結果
    with open('models/track_specific_v3.8.1_summary.json', 'r') as f:
        v381_results = json.load(f)
    
    # 比較各賽道
    comparison = []
    for track in v38_results['tracks']:
        v38_mae = v38_results['tracks'][track]['cv_mae']
        v381_mae = v381_results['tracks'][track]['cv_mae']
        
        improvement = (v38_mae - v381_mae) / v38_mae * 100
        
        comparison.append({
            'track': track,
            'v38_mae': v38_mae,
            'v381_mae': v381_mae,
            'improvement_%': improvement
        })
    
    # 輸出報告
    print("\n" + "="*80)
    print("v3.8 vs v3.8.1 性能比較報告")
    print("="*80)
    
    df_comp = pd.DataFrame(comparison)
    print(df_comp.to_string(index=False))
    
    print(f"\n平均改善: {df_comp['improvement_%'].mean():.2f}%")
    
    return df_comp
```

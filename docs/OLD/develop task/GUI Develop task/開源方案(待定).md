#### 2.1 添加本地緩存管理器
**新檔案**: `modules/gui/base/data_cache_manager.py`

```python
import json
import os
from datetime import datetime, timedelta
from pathlib import Path

class DataCacheManager:
    """數據緩存管理器"""
    
    CACHE_DIR = Path("cache/gui_cache")
    CACHE_EXPIRY = timedelta(minutes=15)  # 15 分鐘過期
    
    @classmethod
    def get_cache_key(cls, analysis_type: str, year: str, race: str, session: str) -> str:
        """生成緩存鍵"""
        return f"{analysis_type}_{year}_{race}_{session}"
    
    @classmethod
    def load_from_cache(cls, cache_key: str) -> dict:
        """從緩存載入數據"""
        cache_file = cls.CACHE_DIR / f"{cache_key}.json"
        
        if not cache_file.exists():
            return None
        
        # 檢查緩存是否過期
        file_time = datetime.fromtimestamp(cache_file.stat().st_mtime)
        if datetime.now() - file_time > cls.CACHE_EXPIRY:
            print(f"[CACHE] 緩存已過期: {cache_key}")
            return None
        
        # 載入緩存
        try:
            with open(cache_file, 'r', encoding='utf-8') as f:
                data = json.load(f)
            
            age = datetime.now() - file_time
            print(f"[CACHE] ✅ 載入緩存: {cache_key} (年齡: {age.seconds // 60} 分鐘)")
            return data
            
        except Exception as e:
            print(f"[ERROR] [CACHE] 載入失敗: {e}")
            return None
    
    @classmethod
    def save_to_cache(cls, cache_key: str, data: dict):
        """保存數據到緩存"""
        try:
            cls.CACHE_DIR.mkdir(parents=True, exist_ok=True)
            cache_file = cls.CACHE_DIR / f"{cache_key}.json"
            
            with open(cache_file, 'w', encoding='utf-8') as f:
                json.dump(data, f, ensure_ascii=False, indent=2)
            
            print(f"[CACHE] ✅ 保存緩存: {cache_key}")
            
        except Exception as e:
            print(f"[ERROR] [CACHE] 保存失敗: {e}")
    
    @classmethod
    def clear_all_cache(cls):
        """清除所有緩存"""
        try:
            import shutil
            if cls.CACHE_DIR.exists():
                shutil.rmtree(cls.CACHE_DIR)
                print("[CACHE] 🗑️ 已清除所有緩存")
        except Exception as e:
            print(f"[ERROR] [CACHE] 清除失敗: {e}")
```

```python
def _load_data_with_current_parameters(self):
```

```python
from abc import ABC, abstractmethod
from typing import Optional, Dict

class APIClientInterface(ABC):
    """API 客戶端接口（開源版本）"""
    
    @abstractmethod
    def get_analysis_data(self, function_id: int, **params) -> Dict:
        """
        獲取分析數據
        
        Args:
            function_id: 分析功能 ID
            **params: 分析參數（year, race, session 等）
            
        Returns:
            Dict: 分析結果數據
            
        Raises:
            APIError: API 請求失敗
        """
        pass
    
    @abstractmethod
    def check_health(self) -> bool:
        """檢查 API 服務器健康狀態"""
        pass

class PublicAPIClient(APIClientInterface):
    """公開 API 客戶端實現"""
    
    def __init__(self, base_url: str = "https://localhost:8000"):
        self.base_url = base_url
        # ✅ 僅實現公開 API 調用邏輯
```

```python
import json
from pathlib import Path
from typing import Optional

class GUIConfig:
    """GUI 配置管理（開源版）"""
    
    DEFAULT_CONFIG = {
        "api": {
            "base_url": "https://localhost:8000",
            "timeout": 30,
            "retry_count": 3
        },
        "cache": {
            "enabled": True,
            "ttl_minutes": 15,
            "path": str(Path.home() / ".f1t-gui" / "cache")
        },
        "ui": {
            "theme": "dark",
            "language": "zh-TW"
        }
    }
    
    @classmethod
    def load_user_config(cls) -> dict:
        """載入用戶配置"""
        config_file = Path.home() / ".f1t-gui" / "config.json"
        if config_file.exists():
            with open(config_file, 'r', encoding='utf-8') as f:
                return json.load(f)
        return cls.DEFAULT_CONFIG
    
    @classmethod
    def save_user_config(cls, config: dict):
        """保存用戶配置"""
        config_file = Path.home() / ".f1t-gui" / "config.json"
        config_file.parent.mkdir(parents=True, exist_ok=True)
        with open(config_file, 'w', encoding='utf-8') as f:
            json.dump(config, f, indent=2, ensure_ascii=False)
```

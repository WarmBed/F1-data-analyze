## 📋 開發核心原則遵循

### ✅ 核心原則檢查清單
- [x] **模組化設計** - 獨立的GUI模組，可單獨測試和維護
- [x] **數據優先** - 優先使用JSON緩存，無數據時自動調用CLI
- [x] **用戶體驗** - MDI子視窗，支援同步和異步操作
- [x] **錯誤處理** - 完整的異常處理和狀態反饋機制
- [x] **性能考量** - 背景執行CLI分析，避免GUI凍結
- [x] **代碼複用** - 使用現有PopoutSubWindow和MDI架構
- [x] **一致性** - 遵循現有GUI風格和命名規範

### 🎯 開發目標
1. 在MDI子視窗中以表格形式顯示車手最快進站時間排行榜
2. 支援JSON數據讀取和CLI後端分析調用
3. 與主視窗參數同步（年份、賽事、賽段）
4. 提供完整的錯誤處理和使用者回饋
5. 支援數據刷新、匯出等進階功能

## 🔧 MDI子視窗技術架構

### 📋 核心類別結構
```
PitstopRankingSubWindow (繼承自 PopoutSubWindow)
├── 🎛️ 初始化與配置
│   ├── __init__(self, parent, year, race, session)
│   ├── setup_ui(self) - 建立使用者界面
│   └── setup_connections(self) - 連接信號槽
│
├── 📊 數據管理
│   ├── data_manager: PitstopDataManager
│   ├── load_data(self) - 載入排行榜數據
│   └── refresh_data(self) - 刷新數據內容
│
├── 🎨 UI元件
│   ├── table_widget: QTableWidget - 排行榜表格
│   ├── status_bar: QStatusBar - 狀態列
│   └── toolbar: QToolBar - 工具列
│
├── 🔄 信號處理
│   ├── parameter_update_received(year, race, session)
│   ├── data_loaded(ranking_data)
│   └── error_occurred(error_message)
│
└── 📤 操作功能
    ├── export_data(self) - 匯出排行榜數據
    ├── show_settings(self) - 顯示設定對話框
    └── close_window(self) - 關閉視窗
```

### 🗂️ PitstopDataManager 數據管理器
```
PitstopDataManager
├── 📁 檔案操作
│   ├── check_json_cache(year, race, session) -> bool
│   ├── load_json_data(file_path) -> dict
│   └── monitor_file_changes(file_path)
│
├── 🔄 CLI備援機制
│   ├── execute_cli_analysis(year, race, session) -> bool
│   ├── wait_for_cli_completion(timeout=60) -> bool
│   └── handle_cli_error(error_info)
│
├── 📊 數據處理
│   ├── parse_ranking_data(raw_data) -> List[Dict]
│   ├── format_time_display(time_seconds) -> str
│   ├── calculate_time_differences(ranking_data)
│   └── sort_by_fastest_time(ranking_data)
│
└── 🎯 狀態管理
    ├── current_state: DataState (LOADING, READY, ERROR)
    ├── loading_progress: int (0-100)
    └── error_info: Dict
```

class PitstopRankingSubWindowManager:
    """進站排行榜 MDI 子視窗管理器"""
    
    def __init__(self, main_window):
        self.main_window = main_window
        self.active_windows = {}  # 存儲活動視窗

    def create_pitstop_ranking_window(self, year=None, race=None, session=None):
        """創建新的進站排行榜視窗"""
        
        # 獲取參數（使用傳入參數或主視窗當前參數）
        current_year = year or self.main_window.get_current_year()
        current_race = race or self.main_window.get_current_race() 
        current_session = session or self.main_window.get_current_session()
        
        # 生成唯一窗口ID
        self.window_counter += 1
        window_id = f"pitstop_ranking_{self.window_counter}"
        
        # 創建子視窗
        sub_window = PitstopRankingSubWindow(
            parent_mdi=self.main_window.get_current_mdi_area(),
            window_id=window_id,
            initial_params={
                'year': current_year,
                'race': current_race,
                'session': current_session
            }
        )
        
        # 註冊視窗
        self.active_windows[window_id] = sub_window
        
        # 連接關閉信號
        sub_window.window_closed.connect(
            lambda: self.on_window_closed(window_id)
        )
        
        # 添加到MDI區域
        mdi_area = self.main_window.get_current_mdi_area()
        mdi_area.addSubWindow(sub_window)
        
        # 計算視窗位置（避免重疊）
        position = self.calculate_window_position()
        sub_window.move(position['x'], position['y'])
        
        # 顯示視窗
        sub_window.show()
        
        print(f"[CREATE] 創建進站排行榜視窗: {window_id} ({current_year} {current_race} {current_session})")
        
        return sub_window
    
    def calculate_window_position(self):
        """計算新視窗位置，避免重疊"""
        offset_x = (self.window_counter % 5) * 30
        offset_y = ((self.window_counter - 1) // 5) * 30
        
        return {
            'x': 20 + offset_x,
            'y': 20 + offset_y
        }
    
    def on_window_closed(self, window_id):
        """處理視窗關閉事件"""
        if window_id in self.active_windows:
            del self.active_windows[window_id]
            print(f"[CLOSE] 關閉進站排行榜視窗: {window_id}")
    
    def get_active_windows(self):
        """獲取所有活動視窗"""
        return list(self.active_windows.values())
    
    def close_all_windows(self):
        """關閉所有進站排行榜視窗"""
        for window in self.active_windows.values():
            window.close()
        self.active_windows.clear()
        print("[CLOSE_ALL] 關閉所有進站排行榜視窗")

        """設定視窗狀態"""
        if new_state not in self.STATES:
            raise ValueError(f"無效狀態: {new_state}")

        self.state_history.append({
            'from': self.current_state,

            'timestamp': datetime.now(),
            'message': message
        })
        
        # 更新當前狀態
        old_state = self.current_state
        self.current_state = new_state
        
        # 更新UI狀態
        self.update_ui_for_state(new_state, message)
        
        print(f"[STATE] 視窗狀態變更: {old_state} -> {new_state} ({message or ''})")

            widget.disable_controls()
            
        elif state == 'LOADING_JSON':

class PitstopRankingDataBinding:
    """進站排行榜數據綁定管理"""
    
    def __init__(self, window):
        self.window = window
        self.widget = window.ranking_widget
        self.current_data = []
        self.current_stats = {}
        self.data_version = 0
        
    def bind_ranking_data(self, ranking_data, stats_data=None):
        """綁定排行榜數據到UI組件"""
        try:
            # 驗證數據格式
            if not self.validate_ranking_data(ranking_data):
                raise ValueError("排行榜數據格式無效")
            
            # 更新數據版本
            self.data_version += 1
            self.current_data = ranking_data
            
            # 計算統計信息（如果未提供）
            if stats_data is None:
                stats_data = self.calculate_stats(ranking_data)
            self.current_stats = stats_data
            
            # 綁定到表格
            self.bind_table_data(ranking_data)
            
            # 綁定到統計面板
            self.bind_stats_data(stats_data)
            
            # 更新狀態
            self.window.state_manager.set_state('READY', f"已載入 {len(ranking_data)} 位車手數據")
            
            print(f"[BIND] 數據綁定完成: 版本 {self.data_version}, {len(ranking_data)} 筆記錄")
            
        except Exception as e:

    def bind_table_data(self, ranking_data):

            diff_item = QTableWidgetItem(driver_data.get('time_diff', '--'))

    def calculate_stats(self, ranking_data):
        """計算統計信息"""
        if not ranking_data:
            return {
                'total_drivers': 0,
                'fastest_time': 0,
                'slowest_time': 0,
                'average_time': 0,
                'time_range': 0
            }
        
        times = [item['fastest_time'] for item in ranking_data if item.get('fastest_time', 0) > 0]
        
        return {
            'total_drivers': len(ranking_data),
            'fastest_time': min(times) if times else 0,
            'slowest_time': max(times) if times else 0,
            'average_time': sum(times) / len(times) if times else 0,
            'time_range': max(times) - min(times) if times else 0
        }
    
    def validate_ranking_data(self, data):
        """驗證排行榜數據格式"""
        if not isinstance(data, list):
            return False
        
        required_fields = ['driver', 'team', 'fastest_time']
        
        for item in data:
            if not isinstance(item, dict):
                return False
            
            for field in required_fields:
                if field not in item:
                    return False
        
        return True
    
    def get_export_data(self, format_type='dict'):
        """獲取可匯出的數據"""
        if format_type == 'dict':
            return {
                'ranking_data': self.current_data,
                'stats_data': self.current_stats,
                'data_version': self.data_version,
                'export_timestamp': datetime.now().isoformat()
            }
        elif format_type == 'csv':
            return self.convert_to_csv()
        elif format_type == 'json':
            return json.dumps(self.get_export_data('dict'), indent=2, ensure_ascii=False)
        else:
            raise ValueError(f"不支援的匯出格式: {format_type}")
    
    def convert_to_csv(self):
        """轉換為CSV格式"""
        import csv
        import io
        
        output = io.StringIO()
        writer = csv.writer(output)
        
        # 寫入標題
        writer.writerow(['排名', '車手', '車隊', '最快時間(秒)', '與第一名差距', '圈數', '賽段時間'])
        
        # 寫入數據
        for item in self.current_data:
            writer.writerow([
                item.get('rank', ''),
                item.get('driver', ''),
                item.get('team', ''),
                item.get('fastest_time', ''),
                item.get('time_diff', ''),
                item.get('lap_number', ''),
                item.get('session_time', '')
            ])
        
        return output.getvalue()

        """處理參數同步"""
        print(f"[EVENT] 處理參數同步: {param_type} = {value}")
        
        # 更新視窗參數
        if param_type == 'year':
            self.widget.current_year = value
        elif param_type == 'race':
            self.widget.current_race = value
        elif param_type == 'session':
            self.widget.current_session = value
        
        # 更新視窗標題
        self.update_window_title()
        
        # 重新載入數據
        self.reload_data_async()
    
    def clear_current_data(self):
        """清除當前數據"""
        self.widget.ranking_table.setRowCount(0)
        self.widget.total_drivers_label.setText("總車手: --")
        self.widget.fastest_time_label.setText("最快: --")
        self.widget.slowest_time_label.setText("最慢: --")
        self.widget.time_range_label.setText("範圍: --")
        self.widget.data_source_label.setText("")
    
    def reload_data_async(self):
        """異步重新載入數據"""
        def load_data():
            try:
                # 嘗試載入JSON
                result = self.widget.data_manager.load_ranking_data(
                    self.widget.current_year,
                    self.widget.current_race,
                    self.widget.current_session
                )
                
                if result['success']:
                    # JSON載入成功
                    QTimer.singleShot(0, lambda: self.on_data_loaded(result))
                else:
                    # 需要CLI分析
                    QTimer.singleShot(0, lambda: self.start_cli_analysis())
                    
            except Exception as e:
                error_msg = f"數據載入失敗: {str(e)}"
                QTimer.singleShot(0, lambda: self.on_data_error(error_msg))
        
        # 在背景線程執行
        import threading
        thread = threading.Thread(target=load_data)
        thread.daemon = True
        thread.start()
    
    def start_cli_analysis(self):
        """開始CLI分析"""
        self.state_manager.set_state('LOADING_CLI')
        
        def on_cli_complete(result):
            if result['success']:
                # CLI成功，重新載入JSON
                self.reload_data_async()
            else:
                error_msg = f"CLI分析失敗: {result.get('message', '未知錯誤')}"
                self.state_manager.set_state('ERROR', error_msg)
        
        # 執行異步CLI分析
        self.widget.cli_manager.execute_async_analysis(
            self.widget.current_year,
            self.widget.current_race,
            self.widget.current_session,
            on_cli_complete
        )
    
    def on_data_loaded(self, result):
        """數據載入完成回調"""
        self.data_binding.bind_ranking_data(
            result['data'],
            {
                'total_drivers': result['total_drivers'],
                'fastest_time': result['fastest_time'],
                'slowest_time': result['slowest_time'],
                'cache_used': result.get('cache_used', False)
            }
        )
    
    def on_data_error(self, error_msg):
        """數據載入錯誤回調"""
        self.state_manager.set_state('ERROR', error_msg)
    
    def generate_export_filename(self, format_type):
        """生成匯出檔案名稱"""
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        base_name = f"pitstop_ranking_{self.widget.current_year}_{self.widget.current_race}_{self.widget.current_session}_{timestamp}"
        
        if format_type == 'csv':
            return f"{base_name}.csv"
        elif format_type == 'json':
            return f"{base_name}.json"
        elif format_type == 'excel':
            return f"{base_name}.xlsx"
        else:
            return f"{base_name}.txt"
    
    def update_window_title(self):
        """更新視窗標題"""
        new_title = f"進站排行榜_{self.widget.current_year}_{self.widget.current_race}_{self.widget.current_session}"
        self.window.setWindowTitle(new_title)
